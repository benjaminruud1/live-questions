    function addQuestionToScreen(text, time) {
      const card = document.createElement('div');
      card.className = 'card';
      card.innerHTML = `
        <div class="anon-tag">👤 Anonymous</div>
        <div class="question-text">${text}</div>
        <div class="time">${new Date(time).toLocaleTimeString()}</div>
      `;
      feed.prepend(card);
    }

    // Subscribe to live database inserts
    supabaseClient
      .channel('public:questions')
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'questions' }, payload => {
        addQuestionToScreen(payload.new.question_text, payload.new.created_at);
      })
      .subscribe();
