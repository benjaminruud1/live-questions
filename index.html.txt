    const SUPABASE_URL = "https://hupuixedfygdlglqhypz.supabase.co";
    const SUPABASE_ANON_KEY = "sb_publishable_pRyCA4Y6FakZLB6w7MrsFw_arG-ERF2";

    const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    async function sendQuestion() {
      const input = document.getElementById('questionInput');
      const status = document.getElementById('status');
      const text = input.value.trim();

      if (!text) return;

      status.style.color = "#666";
      status.textContent = "Sending...";
      
      const { error } = await supabaseClient.from('questions').insert([{ question_text: text }]);

      if (error) {
        status.style.color = "#d93025";
        status.textContent = "Error sending question. Please try again.";
      } else {
        status.style.color = "#137333";
        status.textContent = "✓ Question sent anonymously!";
        input.value = "";
        setTimeout(() => { status.textContent = ""; }, 3000);
      }
    }
